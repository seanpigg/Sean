""" desk.py — Bank Universe landing page. Drop this file next to app.py, then make two small changes in app.py: 1. add to the imports at the top: import desk 2. replace the whole existing home() route with: @app.route("/", methods=["GET"]) def home(): return desk.render_home() That is the entire integration. Nothing else in app.py changes; /analyze, /pdf, /settings, /validate and results.html are untouched. WHAT THIS DOES The old landing page was a setup form. This one runs the screen on arrival: the full universe scored for the newest quarter, ranked by composite, with a band showing what moved since the prior quarter. The bank/quarter picker survives as a "Custom screen" drawer that still POSTs to /analyze unchanged. PERFORMANCE Scoring the full universe twice (current + prior quarter) is memoized below on (as_of, prior, settings-fingerprint). The first hit after a restart or a settings change pays full cost; everything after is instant. Filtering happens on the cached rows, so changing a dropdown never re-scores. If the first load is too slow for comfort, lower PRESCREEN_N or set COMPARE_PRIOR = False to skip prior-quarter scoring (you lose the movers band and the top-decile entrants, and the page still works). """ import analysis import config import reps # How many top-composite banks the ledger works from. Filters refine this list. PRESCREEN_N = 40 # Score the prior quarter too, for q/q deltas. Set False to halve cold-load cost. COMPARE_PRIOR = True # A bank is "in the top decile" at or above this fraction of the universe. TOP_DECILE_FRACTION = 0.10 ASSET_BANDS = [ ("0-250000", "Under $250M", 0, 250_000), ("250000-500000", "$250M – $500M", 250_000, 500_000), ("500000-1000000", "$500M – $1B", 500_000, 1_000_000), ("1000000-3000000", "$1B – $3B", 1_000_000, 3_000_000), ("3000000-10000000", "$3B – $10B", 3_000_000, 10_000_000), ("10000000-", "Over $10B", 10_000_000, None), ] # -------------------------------------------------------------------------- # small helpers — rows may be dicts or objects, so never index them directly # -------------------------------------------------------------------------- def g(row, name, default=None): if row is None: return default if isinstance(row, dict): return row.get(name, default) return getattr(row, name, default) def put(row, name, value): if isinstance(row, dict): row[name] = value else: setattr(row, name, value) def qkey(q): """Sortable integer for a quarter string like '2026Q1'. Matches app.py.""" try: return int(q[:4]) * 10 + int(q[-1]) except (ValueError, IndexError, TypeError): return 0 def band_of(assets_raw): if assets_raw in (None, ""): return None try: a = float(assets_raw) except (TypeError, ValueError): return None for key, _label, lo, hi in ASSET_BANDS: if a >= lo and (hi is None or a < hi): return key return None def fmt_assets(raw): """Workbook assets are in $000. Render as $431M / $1.1B.""" if raw in (None, ""): return None try: a = float(raw) except (TypeError, ValueError): return None if a >= 1_000_000: return "$%.1fB" % (a / 1_000_000) return "$%dM" % round(a / 1_000) def rep_key(row): rep = g(row, "rep") if not rep: return "__unassigned__" if g(rep, "unassigned"): return "__unassigned__" return g(rep, "name") or "__unassigned__" # -------------------------------------------------------------------------- # cached full-universe screen # -------------------------------------------------------------------------- _cache = {} def _fingerprint(settings): """Any setting that changes scores must invalidate the cache.""" return ( settings.get("MATERIALITY_PERCENTILE"), settings.get("SIGNAL_C_REWARD_DECLINE"), tuple(sorted((settings.get("SIGNAL_WEIGHTS") or {}).items())), ) def screen_universe(bank_keys, as_of, prior, settings): """Score every bank for as_of (and prior, for deltas). Memoized.""" ck = (as_of, prior, _fingerprint(settings), len(bank_keys)) if ck in _cache: return _cache[ck] rows = analysis.run_analysis( banks=bank_keys, quarters=[as_of], rep_filter="__all__" ) or [] prior_by_key = {} if prior and COMPARE_PRIOR: try: for r in analysis.run_analysis( banks=bank_keys, quarters=[prior], rep_filter="__all__" ) or []: prior_by_key[g(r, "key")] = r except Exception: # A missing or malformed back-quarter must never break the page. prior_by_key = {} for r in rows: p = prior_by_key.get(g(r, "key")) pc = g(p, "composite") if p else None cc = g(r, "composite") put(r, "prev_composite", pc) put(r, "prev_rank", g(p, "universe_rank") if p else None) put(r, "delta", round(cc - pc) if (pc is not None and cc is not None) else None) for letter in ("a", "b", "c"): ps = g(g(p, "signal_" + letter), "score") if p else None put(r, "prev_signal_" + letter, ps) _cache[ck] = rows return rows def invalidate(): """Called from /refresh and after a settings save.""" _cache.clear() # -------------------------------------------------------------------------- # what changed since last quarter # -------------------------------------------------------------------------- SIGNAL_NAMES = {"a": "Idle liquidity", "b": "Underwater book", "c": "Net income q/q"} def describe_driver(row): """ One short clause naming the signal that moved most. Deterministic on purpose — this is a scan aid on a dense page, so no LLM call. """ best = None for letter in ("a", "b", "c"): now = g(g(row, "signal_" + letter), "score") was = g(row, "prev_signal_" + letter) if now is None or was is None: continue change = now - was if best is None or abs(change) > abs(best[1]): best = (letter, change) if not best or round(abs(best[1])) == 0: return "Composite re-ranked q/q" letter, change = best return "%s %s %d pts" % ( SIGNAL_NAMES[letter], "up" if change > 0 else "down", round(abs(change)) ) def build_movers(rows, limit=5): scored = [r for r in rows if g(r, "delta") is not None] scored.sort(key=lambda r: abs(g(r, "delta")), reverse=True) return [ { "key": g(r, "key"), "bank": g(r, "bank"), "delta": g(r, "delta"), "composite": g(r, "composite"), "prev_composite": g(r, "prev_composite"), "driver": describe_driver(r), } for r in scored[:limit] if g(r, "delta") != 0 ] def build_entrants(rows, universe_total, limit=4): cutoff = max(1, int((universe_total or 0) * TOP_DECILE_FRACTION)) out = [ r for r in rows if g(r, "universe_rank") and g(r, "universe_rank") <= cutoff and g(r, "prev_rank") and g(r, "prev_rank") > cutoff ] out.sort(key=lambda r: g(r, "universe_rank")) return out[:limit], cutoff # -------------------------------------------------------------------------- # the view # -------------------------------------------------------------------------- def render_home(): from flask import render_template, request settings = config.load_settings() threshold = settings.get("MATERIALITY_PERCENTILE", 70) banks, quarters = analysis.get_available_options() quarters = sorted(quarters or [], key=qkey, reverse=True) as_of = quarters[0] if quarters else None prior = quarters[1] if len(quarters) > 1 else None # Assets come off the bank list, which already carries assets_raw. assets_by_key = {} norm_banks = [] for b in (banks or []): key = g(b, "key") or g(b, "value") or str(b) raw = g(b, "assets_raw") assets_by_key[key] = raw norm_banks.append({ "key": key, "label": g(b, "label") or g(b, "bank") or str(b), "assets_fmt": fmt_assets(raw), }) rows = [] if as_of: rows = screen_universe([b["key"] for b in norm_banks], as_of, prior, settings) universe_total = 0 for r in rows: ut = g(r, "universe_total") if ut: universe_total = ut break if not universe_total: universe_total = len(rows) movers = build_movers(rows) entrants_rows, top_decile_rank = build_entrants(rows, universe_total) entrants = [ { "key": g(r, "key"), "bank": g(r, "bank"), "rep": g(r, "rep"), "universe_rank": g(r, "universe_rank"), "prev_rank": g(r, "prev_rank"), } for r in entrants_rows ] prescreen = sorted( [r for r in rows if g(r, "composite") is not None], key=lambda r: g(r, "composite"), reverse=True, )[:PRESCREEN_N] for r in prescreen: raw = assets_by_key.get(g(r, "key")) put(r, "assets_raw", raw) put(r, "assets_fmt", fmt_assets(raw)) q = (request.args.get("q") or "").strip().lower() rep_filter = request.args.get("rep_filter") or "__all__" band = request.args.get("band") or "__all__" def matches(r): if rep_filter != "__all__" and rep_key(r) != rep_filter: return False if band != "__all__" and band_of(g(r, "assets_raw")) != band: return False if q: hay = "%s %s" % (g(r, "bank") or "", g(r, "location") or "") if q not in hay.lower(): return False return True visible = [r for r in prescreen if matches(r)] # Co-filtered dropdowns: each one's counts reflect the other's selection, # so the desk can never pick a combination that returns nothing. rep_pool = [ r for r in prescreen if band == "__all__" or band_of(g(r, "assets_raw")) == band ] rep_counts = {} for r in rep_pool: k = rep_key(r) rep_counts[k] = rep_counts.get(k, 0) + 1 label_for = {o["value"]: o["label"] for o in reps.rep_options()} rep_options = [{"value": "__all__", "label": "All reps"}] for k in sorted(rep_counts, key=lambda x: (x == "__unassigned__", x)): base = label_for.get(k, "Unassigned" if k == "__unassigned__" else k) rep_options.append({"value": k, "label": "%s (%d)" % (base, rep_counts[k])}) if rep_filter != "__all__" and rep_filter not in rep_counts: rep_options.append({ "value": rep_filter, "label": "%s (0)" % label_for.get(rep_filter, rep_filter), }) band_pool = [ r for r in prescreen if rep_filter == "__all__" or rep_key(r) == rep_filter ] band_counts = {} for r in band_pool: b = band_of(g(r, "assets_raw")) if b: band_counts[b] = band_counts.get(b, 0) + 1 band_options = [{"value": "__all__", "label": "All sizes"}] for key, label, _lo, _hi in ASSET_BANDS: n = band_counts.get(key, 0) if n > 0 or key == band: band_options.append({"value": key, "label": "%s (%d)" % (label, n)}) cache = analysis.cache_status() or {} return render_template( "index.html", active_nav="home", as_of=as_of or "—", prior=prior, quarters=quarters, banks=norm_banks, universe=visible, universe_total=universe_total, prescreen_total=len(prescreen), movers=movers, entrants=entrants, top_decile_rank=top_decile_rank, threshold=threshold, above_count=sum( 1 for r in prescreen if (g(r, "composite") or 0) >= threshold ), rep_options=rep_options, band_options=band_options, rep_filter=rep_filter, band=band, q=request.args.get("q") or "", data_dir=config.DATA_DIR, cache=cache, rep_status=reps.status(), freshness="fresh" if cache.get("workbooks_cached") else "no workbook cached", )
